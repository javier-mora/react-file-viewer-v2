## Overview

`react-file-viewer-v2` is a browser-first file preview component for React applications.
It accepts a `Blob`, resolves the requested viewer from `fileType`, and renders a focused preview surface for images, PDF files, spreadsheets, Office documents, videos, and custom unsupported states.

The examples below are intentionally compact.
They are designed to validate the real integration flow instead of presenting a marketing page inside Storybook docs.

## Implementation

### Load a bundled sample

```tsx
import { useState } from "react";
import { FileViewer } from "react-file-viewer-v2";

const sample = new Blob(["..."], { type: "application/pdf" });

export function BundledPreview() {
  return <FileViewer file={sample} fileType="pdf" />;
}
```

### Preview a CSV with an explicit delimiter

```tsx
import { FileViewer } from "react-file-viewer-v2";

export function CsvPreview({ file }: { file: Blob }) {
  return <FileViewer file={file} fileType="csv" csvDelimiter=";" />;
}
```

### Preview an uploaded file

```tsx
import { useState } from "react";
import { FileViewer } from "react-file-viewer-v2";

export function UploadPreview() {
  const [file, setFile] = useState<File | null>(null);
  const [fileType, setFileType] = useState("");

  return (
    <>
      <input
        type="file"
        onChange={(event) => {
          const selected = event.target.files?.[0] ?? null;
          setFile(selected);
          setFileType(selected?.name.split(".").pop()?.toLowerCase() ?? "");
        }}
      />

      {file ? <FileViewer file={file} fileType={fileType} /> : null}
    </>
  );
}
```

### Provide a custom fallback

```tsx
import { FileViewer } from "react-file-viewer-v2";

export function RestrictedPreview({ file }: { file: Blob }) {
  return (
    <FileViewer
      file={file}
      fileType="pdf"
      omit={["pdf"]}
      unsupportedComponent={<div>PDF preview is disabled for this workspace.</div>}
    />
  );
}
```
